func TestNativeOperatorLifetimeEOFStopsPrivateWorker(t *testing.T) {
	read, write, err := os.Pipe()
	if err != nil {
		t.Fatal(err)
	}
	defer write.Close()
	ctx, closeLifetime, err := nativeOperatorLifetimeFileContext(context.Background(), read)
	if err != nil {
		read.Close()
		t.Fatal(err)
	}
	defer closeLifetime()
	if ctx.Err() != nil {
		t.Fatal("live parent appeared dead")
	}
	if err = write.Close(); err != nil {
		t.Fatal(err)
	}
	select {
	case <-ctx.Done():
	case <-time.After(time.Second):
		t.Fatal("lost parent did not cancel operator worker")
	}
}

func TestNativeOperatorLifetimeCleanupUnblocksPrivateReader(t *testing.T) {
	read, write, err := os.Pipe()
	if err != nil {
		t.Fatal(err)
	}
	defer write.Close()
	_, closeLifetime, err := nativeOperatorLifetimeFileContext(context.Background(), read)
	if err != nil {
		read.Close()
		t.Fatal(err)
	}
	done := make(chan struct{})
	go func() { closeLifetime(); close(done) }()
	select {
	case <-done:
	case <-time.After(time.Second):
		t.Fatal("worker cleanup left lifetime reader blocked with parent alive")
	}
}

func TestNativeOperatorLifetimeRejectsRegularFileAndMalformedHandle(t *testing.T) {
	f, err := os.Create(filepath.Join(t.TempDir(), "not-pipe"))
	if err != nil {
		t.Fatal(err)
	}
	if _, _, err = nativeOperatorLifetimeFileContext(context.Background(), f); err == nil {
		t.Fatal("regular file became parent lifetime lease")
	}
	for _, value := range []string{"", "0", "03", "not-a-handle"} {
		t.Setenv(platform.NativeBrokerOperatorLifetimeEnv, value)
		if _, _, err = nativeOperatorLifetimeContext(context.Background()); err == nil {
			t.Fatalf("accepted handle %q", value)
		}
	}
}

func TestNativeOperatorSupervisorRefusesCallerArguments(t *testing.T) {
	var out bytes.Buffer
	args := append(cliinvoke.NativeOperatorSupervisorRun(), "--home", "/caller")
	if handled, code := runNativeOperatorSupervisorCommand(args, &out); !handled || code != 1 {
		t.Fatal("arbitrary worker arguments accepted")
	}
}
