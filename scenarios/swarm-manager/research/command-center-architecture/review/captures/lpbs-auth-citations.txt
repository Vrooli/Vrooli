 444  // ValidateServiceToken validates the service-to-service auth token.
 445  // Uses constant-time comparison to prevent timing attacks where an attacker
 446  // could deduce token characters by measuring response times.
 447  func (s *UsageService) ValidateServiceToken(token string) bool {
 448      if s.serviceToken == "" {
 449          // No token configured - reject all
 450          return false
 451      }
 452      // Use constant-time comparison to prevent timing attacks
 453      return subtle.ConstantTimeCompare([]byte(token), []byte(s.serviceToken)) == 1
 454  }
